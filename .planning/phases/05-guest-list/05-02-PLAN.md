**handleToggleInvited function:**
```typescript
function handleToggleInvited(id: string, currentInvited: boolean) {
  startTransition(async () => {
    setOptimisticGuests({ type: "update", guest: { ...optimisticGuests.find(g => g.id === id)!, invited: !currentInvited } });
    const result = await updateGuest(id, { invited: !currentInvited });
    if (result?.error) setError(result.error);
    else router.refresh();
  });
}
```
